class ControladorDigitalPID { constructor(Kp, Ti, Td, Ts, acaoDireta = false, outMin = 0, outMax = 100) { // 3. Parâmetros equivalentes a PID this.Kp = Kp; this.Ti = Ti; this.Td = Td; this.Ts = Ts; // Limites para Anti Reset Wind-up this.outMin = outMin; this.outMax = outMax; // 9. Ação Direta ou Reversa (1 = Reversa/Aquecimento, -1 = Direta/Resfriamento) this.dir = acaoDireta ? -1 : 1; // Memória de estados (Segunda Ordem) this.e_k1 = 0; this.pv_k1 = 0; this.pv_k2 = 0; this.u_k1 = 0; } // Método chamado a cada ciclo de amostragem (Ts) calcular(sp, pv, isAuto, manualMv, spTrackingAtivo = true) { // 4. Transferência Automático / Manual if (!isAuto) { // 6. Função Tracking para SP e VM if (spTrackingAtivo) sp = pv; // SP rastreia a PV em manual let u_manual = Math.max(this.outMin, Math.min(this.outMax, manualMv)); // 5. Função Bumpless: Preparar memória para o retorno ao Automático this.u_k1 = u_manual; this.e_k1 = this.dir * (sp - pv); this.pv_k1 = pv; this.pv_k2 = pv; return { mv: u_manual, novoSp: sp }; // Retorna a saída manual } // --- MODO AUTOMÁTICO (1. Algoritmo Incremental / 2. Segunda Ordem) --- let e_k = this.dir * (sp - pv); // Termo Proporcional let deltaP = this.Kp * (e_k - this.e_k1); // Termo Integral (8. Gera rampa como resposta ao erro em degrau) let deltaI = 0; if (this.Ti > 0) { deltaI = this.Kp * (this.Ts / this.Ti) * e_k; } // Termo Derivativo (10. Aplicada somente à PV) let deltaD = 0; if (this.Td > 0) { deltaD = -this.dir * this.Kp * (this.Td / this.Ts) * (pv - 2 * this.pv_k1 + this.pv_k2); } // Variação total let deltaU = deltaP + deltaI + deltaD; // Saída atualizada let u_k = this.u_k1 + deltaU; // 7. Anti Reset Wind-up (saturação imediata na forma de velocidade) u_k = Math.max(this.outMin, Math.min(this.outMax, u_k)); // Atualização da memória para o próximo ciclo this.e_k1 = e_k; this.pv_k2 = this.pv_k1; this.pv_k1 = pv; this.u_k1 = u_k; return { mv: u_k, novoSp: sp }; } }
